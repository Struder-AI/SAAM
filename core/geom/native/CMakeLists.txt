cmake_minimum_required(VERSION 3.22)
project(saam_mesh_repair LANGUAGES CXX)
find_package(CGAL 6.2.1 EXACT REQUIRED)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
include(CGAL_Eigen3_support)
add_executable(saam-mesh-repair mesh-repair.cpp)
target_compile_features(saam-mesh-repair PRIVATE cxx_std_17)
target_link_libraries(saam-mesh-repair PRIVATE CGAL::CGAL CGAL::Eigen3_support)
install(TARGETS saam-mesh-repair RUNTIME DESTINATION .)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/mesh-repair.cpp" WRAPPER_SHA)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build.json" "{\"cgal\":\"6.2.1\",\"sourceSha256\":\"${WRAPPER_SHA}\"}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build.json" DESTINATION .)
